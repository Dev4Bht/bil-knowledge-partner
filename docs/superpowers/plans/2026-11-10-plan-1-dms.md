# Plan 1: BIL Document Management System (DMS) Implementation Plan

> **SUPERSEDED (2026-11-10).** Tasks 1 to 4 were built and tested against SQLite, then retired when the owner supplied `CLAUDE.md` (Postgres, MinIO, Next.js, local models). The SQLite prototype is in git history at commit `3760fd0`. Domain rules from it (folder ACL inheritance, atomic supersession, CID screening, content-addressed storage) carry over to `apps/api`.

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a working DMS where an admin uploads documents into business-line folders with versions and access rights, staff browse them, and a read-only API exposes everything the later knowledge layer needs.

**Architecture:** One FastAPI service. Domain rules (folders, versions, access, audit, personal-data screening) live in a `services/` layer that has no HTTP knowledge; thin routers translate HTTP to services. SQLite holds metadata, a content-addressed folder holds files. A `service` role can only read, which is the read-only boundary the AI agent will use. A plain HTML/JS page is the web UI.

**Tech Stack:** Python 3.12, uv, FastAPI, SQLAlchemy 2 (SQLite), itsdangerous (tokens), python-multipart, pytest, httpx, Docker Compose.

**Spec:** `docs/superpowers/specs/2026-11-10-bil-policy-agent-design.md` (sections 4, 5.1, 6, 7, 8). This is plan 1 of 4. Later plans (not in this document): 2 knowledge layer, 3 agent, 4 chat UI plus English/Dzongkha drafts plus evaluation.

## Global Constraints

- Python 3.12; FastAPI; SQLAlchemy with SQLite; no external services to run.
- Web UI is plain HTML, CSS and JavaScript with **no build step**.
- Run everything with a single `docker compose up`.
- Version statuses are exactly `draft`, `current`, `superseded`. Superseded files are never deleted.
- Business lines are exactly `general_insurance`, `credit`, `pf_gf`, `corporate`.
- Allowed upload types: PDF, DOCX, XLSX, TXT.
- Documents must be unclassified or public-facing. **No customer personal data**; the upload screen says so and plain-text uploads are screened for Bhutanese CID-style 11-digit numbers.
- The knowledge layer and agent may use **only the read-only API** (GET endpoints, `service` role). The `service` role must get 403 on every write.
- Roles: `admin` (writes), `staff` (reads what their groups allow), `service` (reads everything, never writes).
- Test-first. Every task ends with a commit. Commit messages end with the line `Co-Authored-By: Claude Sonnet 5.5 <noreply@anthropic.com>`.
- Development data is **synthetic**. Real documents, databases and indexes are git-ignored (`data/`).
- Do not push to the remote unless the owner asks.

`src/bil/dms/security.py`:

```python
import hashlib
import hmac
import os

from itsdangerous import BadSignature, SignatureExpired, URLSafeTimedSerializer

_ITERATIONS = 120_000


def hash_password(password: str) -> str:
    salt = os.urandom(16)
    digest = hashlib.pbkdf2_hmac("sha256", password.encode(), salt, _ITERATIONS)
    return f"pbkdf2_sha256${_ITERATIONS}${salt.hex()}${digest.hex()}"


def verify_password(password: str, stored: str) -> bool:
    try:
        _algo, iterations, salt_hex, hash_hex = stored.split("$")
        digest = hashlib.pbkdf2_hmac("sha256", password.encode(), bytes.fromhex(salt_hex), int(iterations))
    except ValueError:
        return False
    return hmac.compare_digest(digest.hex(), hash_hex)


def _serializer(secret: str) -> URLSafeTimedSerializer:
    return URLSafeTimedSerializer(secret, salt="bil-dms-token")


def make_token(secret: str, username: str) -> str:
    return _serializer(secret).dumps({"u": username})


def read_token(secret: str, token: str, max_age: int) -> str | None:
    try:
        return _serializer(secret).loads(token, max_age=max_age)["u"]
    except (BadSignature, SignatureExpired, KeyError):
        return None
```

```python
class DocType(str, enum.Enum):
    handbook = "handbook"
    sop = "sop"
    circular = "circular"
    policy_wording = "policy_wording"
    product_catalogue = "product_catalogue"
    form = "form"
    guideline = "guideline"
    training = "training"
    other = "other"


class VersionStatus(str, enum.Enum):
    draft = "draft"
    current = "current"
    superseded = "superseded"


def _enum(enum_cls):
    return SAEnum(enum_cls, native_enum=False, length=32, validate_strings=True)


user_groups = Table(
    "user_groups", Base.metadata,
    Column("user_id", ForeignKey("users.id"), primary_key=True),
    Column("group_id", ForeignKey("groups.id"), primary_key=True),
)

folder_read_groups = Table(
    "folder_read_groups", Base.metadata,
    Column("folder_id", ForeignKey("folders.id"), primary_key=True),
    Column("group_id", ForeignKey("groups.id"), primary_key=True),
)


class Group(Base):
    __tablename__ = "groups"
    id: Mapped[int] = mapped_column(primary_key=True)
    name: Mapped[str] = mapped_column(String(100), unique=True)


class User(Base):
    __tablename__ = "users"
    id: Mapped[int] = mapped_column(primary_key=True)
    username: Mapped[str] = mapped_column(String(100), unique=True)
    full_name: Mapped[str] = mapped_column(String(200))
    password_hash: Mapped[str] = mapped_column(String(300))
    role: Mapped[Role] = mapped_column(_enum(Role))
    groups: Mapped[list[Group]] = relationship(secondary=user_groups)


class Folder(Base):
    __tablename__ = "folders"
    id: Mapped[int] = mapped_column(primary_key=True)
    name: Mapped[str] = mapped_column(String(200))
    parent_id: Mapped[Optional[int]] = mapped_column(ForeignKey("folders.id"), nullable=True)
    parent: Mapped[Optional["Folder"]] = relationship(back_populates="children", remote_side="Folder.id")
    children: Mapped[list["Folder"]] = relationship(back_populates="parent")
    read_groups: Mapped[list[Group]] = relationship(secondary=folder_read_groups)


class Document(Base):
    __tablename__ = "documents"
    id: Mapped[int] = mapped_column(primary_key=True)
    folder_id: Mapped[int] = mapped_column(ForeignKey("folders.id"))

    title: Mapped[str] = mapped_column(String(300))

    versions: Mapped[list["DocumentVersion"]] = relationship(

    target_type: Mapped[str] = mapped_column(String(50))

    detail: Mapped[str] = mapped_column(Text, default="")
```

```python
from .models import Folder, Role, User


def effective_read_groups(folder: Folder) -> list[str]:
    """Groups allowed to read this folder. Inherited from the nearest ancestor that sets any.
    An empty list means every signed-in user may read it."""
    node: Folder | None = folder
    while node is not None:
        if node.read_groups:
            return sorted(g.name for g in node.read_groups)
        node = node.parent
    return []


def can_read_folder(user: User, folder: Folder) -> bool:
    if user.role in (Role.admin, Role.service):
        return True
    required = effective_read_groups(folder)
    if not required:
        return True
    return bool(set(required) & {g.name for g in user.groups})
```

```python
    if group is None:
        group = Group(name=name)
        session.add(group)
        session.flush()

def folder_path(folder: Folder) -> str:
    parts = []
    node: Folder | None = folder
    while node is not None:
        parts.append(node.name)
        node = node.parent
    return "/".join(reversed(parts))


def create_folder(
    session: Session, *, name: str, parent_id: int | None = None,
    read_groups: Iterable[str] = (), actor: str = "system",
) -> Folder:
    name = name.strip()
    if not name or "/" in name:
        raise ValidationFailed("Folder name must not be empty or contain '/'")
    if parent_id is not None and session.get(Folder, parent_id) is None:
        raise NotFoundError(f"Parent folder {parent_id} not found")
    if session.scalar(select(Folder).where(Folder.name == name, Folder.parent_id == parent_id)) is not None:
        raise ConflictError(f"Folder '{name}' already exists here")
    folder = Folder(name=name, parent_id=parent_id, read_groups=[get_or_create_group(session, g) for g in read_groups])
    session.add(folder)
    session.flush()
    audit.record(session, username=actor, action="folder_created", target_type="folder", target_id=folder.id, detail=name)
    session.commit()
    return folder


def get_folder_by_path(session: Session, path: str) -> Folder | None:
    parent_id: int | None = None
    folder: Folder | None = None
    for part in path.split("/"):
        folder = session.scalar(select(Folder).where(Folder.name == part, Folder.parent_id == parent_id))
        if folder is None:
            return None
        parent_id = folder.id
    return folder
```

```python
import hashlib
from pathlib import Path


class FileStorage:
    """Content-addressed file store. Original file names live in the database, never on disk."""

    def __init__(self, root: Path):
        self.root = Path(root)
        self.root.mkdir(parents=True, exist_ok=True)

    def save(self, data: bytes) -> tuple[str, str]:
        sha = hashlib.sha256(data).hexdigest()
        relative = f"{sha[:2]}/{sha}"
        target = self.root / relative
        target.parent.mkdir(parents=True, exist_ok=True)
        if not target.exists():
            target.write_bytes(data)
        return relative, sha

    def path(self, relative: str) -> Path:
        resolved = (self.root / relative).resolve()
        if not resolved.is_relative_to(self.root.resolve()):
            raise ValueError("path escapes storage root")
        return resolved
```

```python
import re

# Bhutanese CID numbers are 11 digits. Longer or shorter digit runs are not matched.
_CID = re.compile(r"(?<!\d)\d{11}(?!\d)")


def scan_text(text: str) -> list[dict]:
    findings = []
    for match in _CID.finditer(text):
        value = match.group()
        findings.append({"kind": "cid_number", "masked": f"{value[:4]}***{value[-2:]}"})
    return findings
```

```python
from datetime import date
from pathlib import Path

from sqlalchemy import select
from sqlalchemy.orm import Session

from ..errors import NotFoundError, ValidationFailed
from ..models import BusinessLine, Change, DocType, Document, DocumentVersion, Folder, User, VersionStatus
from ..permissions import can_read_folder
from ..pii import scan_text
from ..storage import FileStorage
from . import audit

MAX_BYTES = 25 * 1024 * 1024
ALLOWED_EXTENSIONS = {".pdf", ".docx", ".xlsx", ".txt"}


def current_version(document: Document) -> DocumentVersion | None:
    return next((v for v in reversed(document.versions) if v.status == VersionStatus.current), None)


def _validate_upload(filename: str, data: bytes, status: VersionStatus, confirm_no_personal_data: bool) -> None:
    extension = Path(filename).suffix.lower()
    if extension not in ALLOWED_EXTENSIONS:
        raise ValidationFailed(f"File type '{extension}' is not allowed. Use PDF, DOCX, XLSX or TXT.")
    if not data:
        raise ValidationFailed("The file is empty")
    if len(data) > MAX_BYTES:
        raise ValidationFailed(f"The file is larger than {MAX_BYTES // (1024 * 1024)} MB")
    if status not in (VersionStatus.draft, VersionStatus.current):
        raise ValidationFailed("A new version can only be uploaded as draft or current")
    if extension == ".txt" and not confirm_no_personal_data:
        findings = scan_text(data.decode("utf-8", errors="ignore"))
        if findings:
            raise ValidationFailed(
                f"Possible personal data found ({len(findings)} item(s), for example {findings[0]['masked']}). "
                "Remove it, or confirm that the document contains no customer personal data."
            )


def _log_change(session: Session, version: DocumentVersion, kind: str) -> None:
    session.add(Change(version_id=version.id, kind=kind))
    session.flush()


def _supersede_others(session: Session, document: Document, keep: DocumentVersion, actor: str) -> None:
    for other in document.versions:
        if other is not keep and other.status == VersionStatus.current:
            other.status = VersionStatus.superseded
            session.flush()
            _log_change(session, other, "status_changed")
            audit.record(session, username=actor, action="version_superseded", target_type="version", target_id=other.id)


def _new_version(
    session: Session, storage: FileStorage, *, document: Document, user: User, filename: str,
    content_type: str, data: bytes, effective_date: date | None, status: VersionStatus, note: str,
) -> DocumentVersion:
    relative, sha = storage.save(data)
    number = max((v.version_number for v in document.versions), default=0) + 1
    version = DocumentVersion(
        document=document, version_number=number, filename=filename, content_type=content_type,
        size_bytes=len(data), sha256=sha, storage_path=relative, effective_date=effective_date,
        status=status, uploaded_by=user.username, note=note,
    )
    session.add(version)
    session.flush()
    if status == VersionStatus.current:
        _supersede_others(session, document, version, user.username)
    _log_change(session, version, "created")
    audit.record(session, username=user.username, action="version_uploaded", target_type="version",
                 target_id=version.id, detail=f"{document.title} v{number} {filename}")
    return version


def create_document_with_version(
    session: Session, storage: FileStorage, *, user: User, folder_id: int, title: str,
    business_line: BusinessLine, doc_type: DocType, owner: str, filename: str, content_type: str,
    data: bytes, effective_date: date | None = None, status: VersionStatus = VersionStatus.current,
    note: str = "", confirm_no_personal_data: bool = False,
) -> DocumentVersion:
    title = title.strip()
    if not title:
        raise ValidationFailed("Title must not be empty")
    folder = session.get(Folder, folder_id)
    if folder is None:
        raise NotFoundError(f"Folder {folder_id} not found")
    _validate_upload(filename, data, status, confirm_no_personal_data)
    document = Document(folder=folder, title=title, business_line=business_line, doc_type=doc_type, owner=owner)
    session.add(document)
    session.flush()
    version = _new_version(session, storage, document=document, user=user, filename=filename,
                           content_type=content_type, data=data, effective_date=effective_date,
                           status=status, note=note)
    session.commit()
    return version


def add_version(
    session: Session, storage: FileStorage, *, user: User, document_id: int, filename: str,
    content_type: str, data: bytes, effective_date: date | None = None,
    status: VersionStatus = VersionStatus.current, note: str = "", confirm_no_personal_data: bool = False,
) -> DocumentVersion:
    document = session.get(Document, document_id)
    if document is None:
        raise NotFoundError(f"Document {document_id} not found")
    _validate_upload(filename, data, status, confirm_no_personal_data)
    version = _new_version(session, storage, document=document, user=user, filename=filename,
                           content_type=content_type, data=data, effective_date=effective_date,
                           status=status, note=note)
    session.commit()
    return version


def set_version_status(session: Session, *, user: User, version_id: int, status: VersionStatus) -> DocumentVersion:
    version = session.get(DocumentVersion, version_id)
    if version is None:
        raise NotFoundError(f"Version {version_id} not found")
    if status == VersionStatus.draft:
        raise ValidationFailed("A version cannot be moved back to draft")
    if version.status == status:
        return version
    version.status = status
    session.flush()
    if status == VersionStatus.current:
        _supersede_others(session, version.document, version, user.username)
    _log_change(session, version, "status_changed")
    audit.record(session, username=user.username, action="status_changed", target_type="version",
                 target_id=version.id, detail=status.value)
    session.commit()
    return version


def list_documents(
    session: Session, user: User, *, folder_id: int | None = None, business_line: BusinessLine | None = None,
) -> list[Document]:
    query = select(Document).order_by(Document.title)
    if folder_id is not None:
        query = query.where(Document.folder_id == folder_id)
    if business_line is not None:
        query = query.where(Document.business_line == business_line)
    return [d for d in session.scalars(query) if can_read_folder(user, d.folder)]


def get_document_for_read(session: Session, user: User, document_id: int) -> Document:
    document = session.get(Document, document_id)
    if document is None or not can_read_folder(user, document.folder):
        raise NotFoundError(f"Document {document_id} not found")
    return document


def get_version_for_read(session: Session, user: User, version_id: int) -> DocumentVersion:
    version = session.get(DocumentVersion, version_id)
    if version is None or not can_read_folder(user, version.document.folder):
        raise NotFoundError(f"Version {version_id} not found")
    return version


def list_changes(session: Session, user: User, *, since: int = 0, limit: int = 500) -> list[Change]:
    query = select(Change).where(Change.seq > since).order_by(Change.seq).limit(limit)
    return [c for c in session.scalars(query) if can_read_folder(user, c.version.document.folder)]
```

```python
from fastapi import Depends, Header, HTTPException, Request

from ..models import Role, User
from ..security import read_token


def get_session(request: Request):
    with request.app.state.session_factory() as session:
        yield session


def get_storage(request: Request):
    return request.app.state.storage


def get_current_user(
    request: Request, session=Depends(get_session), authorization: str | None = Header(default=None),
) -> User:
    if not authorization or not authorization.lower().startswith("bearer "):
        raise HTTPException(401, "Missing bearer token")
    settings = request.app.state.settings
    username = read_token(settings.secret_key, authorization[7:], settings.token_max_age_seconds)
    user = session.query(User).filter_by(username=username).one_or_none() if username else None
    if user is None:
        raise HTTPException(401, "Invalid or expired token")
    return user


def require_admin(user: User = Depends(get_current_user)) -> User:
    if user.role != Role.admin:
        raise HTTPException(403, "Administrator access required")
    return user
```

```python
from ..models import User
from ..schemas import LoginIn, user_out
from ..security import make_token, verify_password
from .deps import get_current_user, get_session

router = APIRouter()


@router.post("/auth/login")
def login(body: LoginIn, request: Request, session=Depends(get_session)):
    user = session.query(User).filter_by(username=body.username).one_or_none()
    if user is None or not verify_password(body.password, user.password_hash):
        raise HTTPException(401, "Wrong username or password")
    token = make_token(request.app.state.settings.secret_key, user.username)
    return {"token": token, **user_out(user).model_dump()}
```

```python
from pathlib import Path

from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse
from fastapi.staticfiles import StaticFiles

from ..config import Settings, load_settings
from .api import auth, documents, folders, system
from .db import init_db, make_engine, make_session_factory
from .errors import ConflictError, NotFoundError, ValidationFailed
from .storage import FileStorage

WEB_DIR = Path(__file__).parent / "web"


def create_app(settings: Settings | None = None) -> FastAPI:
    settings = settings or load_settings()
    settings.data_dir.mkdir(parents=True, exist_ok=True)
    engine = make_engine(settings.database_url)
    init_db(engine)

    app = FastAPI(title="BIL Document Management System")
    app.state.settings = settings
    app.state.session_factory = make_session_factory(engine)
    app.state.storage = FileStorage(settings.files_dir)

    for exc_type, status in ((NotFoundError, 404), (ConflictError, 409), (ValidationFailed, 422)):
        def handler(request: Request, exc, status=status):
            return JSONResponse({"detail": str(exc)}, status_code=status)
        app.add_exception_handler(exc_type, handler)

    for module in (system, auth, folders, documents):
        app.include_router(module.router, prefix="/api")

    if WEB_DIR.exists():
        app.mount("/", StaticFiles(directory=WEB_DIR, html=True), name="web")
    return app
```

```javascript
  const node = document.createElement(tag);
  for (const [key, value] of Object.entries(attrs || {})) {
    if (key.startsWith("on")) node.addEventListener(key.slice(2), value);
    else if (key === "class") node.className = value;
    else if (value === true) node.setAttribute(key, "");
    else if (value !== false && value != null) node.setAttribute(key, value);
```

```python
def load_sample_corpus(session: Session, storage: FileStorage, admin: User) -> int:
    created = 0
    for sample in SAMPLES:
        folder = get_folder_by_path(session, sample.folder)
        if folder is None:
            raise RuntimeError(f"Folder '{sample.folder}' is missing; run the demo seed first")
        if session.query(Document).filter_by(folder_id=folder.id, title=sample.title).first():
            continue
        first_note, first_text, first_date = sample.versions[0]
        version = docs.create_document_with_version(
            session, storage, user=admin, folder_id=folder.id, title=sample.title,
            business_line=sample.business_line, doc_type=sample.doc_type, owner=sample.owner,
            filename=sample.filename, content_type="text/plain", data=first_text.encode(),
            effective_date=first_date, status=VersionStatus.current, note=first_note,
        )
        for note, text, effective in sample.versions[1:]:
            docs.add_version(session, storage, user=admin, document_id=version.document_id,
                             filename=sample.filename, content_type="text/plain", data=text.encode(),
                             effective_date=effective, note=note)
        created += 1
    return created

        admin = session.query(User).filter_by(username="admin").one()
        count = load_sample_corpus(session, FileStorage(settings.files_dir), admin)
    print(f"Loaded {count} synthetic sample documents")
```
